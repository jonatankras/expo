import React, { useState, useRef } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TouchableOpacity,
  ScrollView,
  TextInput,
  PanResponder,
  Animated,
  SafeAreaView,
  Image,
} from 'react-native';
import { CameraView, useCameraPermissions } from 'expo-camera';
import * as MediaLibrary from 'expo-media-library';
import * as Haptics from 'expo-haptics';
import * as FileSystem from 'expo-file-system';
import * as Sharing from 'expo-sharing';
import * as XLSX from 'xlsx';

const COLS = ['A', 'B', 'C', 'D', 'E'];

interface OcrChip {
  id: string;
  text: string;
  cleanValue: string;
  category: 'code' | 'description' | 'quantity' | 'currency' | 'date';
  x: number;
  y: number;
  w: number;
  h: number;
}

interface CellItem {
  raw: string;
  bold?: boolean;
  italic?: boolean;
  underline?: boolean;
  align?: 'left' | 'center' | 'right';
  format?: 'general' | 'currency' | 'number';
}

const INITIAL_OCR_CHIPS: OcrChip[] = [
  { id: 'c1', text: '98413', cleanValue: '98413', category: 'code', x: 5, y: 22, w: 20, h: 14 },
  { id: 'c2', text: 'Adaptador Inox', cleanValue: 'Adaptador Inox', category: 'description', x: 28, y: 22, w: 40, h: 14 },
  { id: 'c3', text: '24', cleanValue: '24', category: 'quantity', x: 70, y: 22, w: 14, h: 14 },
  { id: 'c4', text: 'R$ 45,90', cleanValue: '45.90', category: 'currency', x: 5, y: 44, w: 28, h: 14 },
  { id: 'c5', text: '98414', cleanValue: '98414', category: 'code', x: 5, y: 66, w: 20, h: 14 },
  { id: 'c6', text: 'Parafuso M12', cleanValue: 'Parafuso M12', category: 'description', x: 28, y: 66, w: 40, h: 14 },
  { id: 'c7', text: '120', cleanValue: '120', category: 'quantity', x: 70, y: 66, w: 14, h: 14 },
];

export default function App() {
  const [cameraPermission, requestCameraPermission] = useCameraPermissions();
  const [mediaPermission, requestMediaPermission] = MediaLibrary.usePermissions();
  const cameraRef = useRef<CameraView>(null);

  // Passo 1: Célula A3 (row 2, col 0) selecionada inicialmente
  const [grid, setGrid] = useState<CellItem[][]>(() => {
    const rows: CellItem[][] = Array.from({ length: 12 }, (_, r) =>
      Array.from({ length: 5 }, (__, c) => ({
        raw: '',
        align: c >= 2 ? 'right' : 'left',
        format: c >= 3 ? 'currency' : c === 2 ? 'number' : 'general',
      }))
    );
    rows[0] = [
      { raw: 'Código', bold: true },
      { raw: 'Descrição', bold: true },
      { raw: 'Qtd', bold: true, align: 'right' },
      { raw: 'Preço Unit.', bold: true, align: 'right' },
      { raw: 'Subtotal', bold: true, align: 'right' },
    ];
    rows[1] = [
      { raw: '98412', bold: true },
      { raw: 'Cabo USB-C 2m' },
      { raw: '15', align: 'right', format: 'number' },
      { raw: '28.50', align: 'right', format: 'currency' },
      { raw: '=C2*D2', align: 'right', format: 'currency', bold: true },
    ];
    for (let r = 2; r < 10; r++) {
      rows[r][4] = { raw: `=C${r + 1}*D${r + 1}`, align: 'right', format: 'currency' };
    }
    return rows;
  });

  const [activeCell, setActiveCell] = useState({ row: 2, col: 0 });
  const [autoAdvance, setAutoAdvance] = useState<'row' | 'col'>('row');
  const [showRibbon, setShowRibbon] = useState(true);
  const [activeTab, setActiveTab] = useState<'coletor' | 'excel' | 'formulas'>('coletor');
  const [torchOn, setTorchOn] = useState(false);

  // Passo 2 e 3: Congelamento da Câmera e OCR
  const [isFrozen, setIsFrozen] = useState(false);
  const [frozenPhotoUri, setFrozenPhotoUri] = useState<string | null>(null);
  const [statusToast, setStatusToast] = useState<string | null>(null);

  const panX = useRef(new Animated.Value(0)).current;

  const showFeedback = (msg: string) => {
    setStatusToast(msg);
    setTimeout(() => setStatusToast(null), 2200);
  };

  const handleFreezeFrame = async () => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);
    if (cameraRef.current && cameraPermission?.granted) {
      try {
        const photo = await cameraRef.current.takePictureAsync({ quality: 0.85, skipProcessing: true });
        if (photo?.uri) setFrozenPhotoUri(photo.uri);
      } catch {}
    }
    setIsFrozen(true);
    showFeedback('Frame congelado · Toque nos dados para inserir');
  };

  // Passo 5: Gestos Horizontais (Swipe Esquerda = Excluir | Swipe Direita = Salvar Galeria)
  const panResponder = useRef(
    PanResponder.create({
      onMoveShouldSetPanResponder: (_, g) => Math.abs(g.dx) > 18 && Math.abs(g.dx) > Math.abs(g.dy),
      onPanResponderMove: Animated.event([null, { dx: panX }], { useNativeDriver: false }),
      onPanResponderRelease: async (_, g) => {
        if (g.dx < -65) {
          Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning);
          setIsFrozen(false);
          setFrozenPhotoUri(null);
          showFeedback('← Captura excluída · Câmera pronta para nova coleta');
        } else if (g.dx > 65) {
          Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
          if (frozenPhotoUri) {
            if (!mediaPermission?.granted) await requestMediaPermission();
            await MediaLibrary.saveToLibraryAsync(frozenPhotoUri);
          }
          showFeedback('→ Imagem salva na Galeria de Fotos!');
        }
        Animated.spring(panX, { toValue: 0, useNativeDriver: false }).start();
      },
    })
  ).current;

  // Passo 4: Inserção com Toque e Auto-Avanço
  const handleSelectOcrChip = (chip: OcrChip) => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    setGrid((prev) => {
      const next = prev.map((r) => r.map((c) => ({ ...c })));
      next[activeCell.row][activeCell.col].raw = chip.cleanValue;
      return next;
    });
    setActiveCell((prev) => {
      if (autoAdvance === 'row') {
        const nextCol = prev.col + 1;
        if (nextCol < 4) return { row: prev.row, col: nextCol };
        return { row: Math.min(11, prev.row + 1), col: 0 };
      }
      return { row: Math.min(11, prev.row + 1), col: prev.col };
    });
  };

  // Passo 6: Barra Vertical Translúcida na Extrema Direita (TAB, ▲, ►, ◄, ▼, ↵)
  const navigateCell = (dir: 'tab' | 'up' | 'right' | 'left' | 'down' | 'enter') => {
    Haptics.selectionAsync();
    setActiveCell((prev) => {
      if (dir === 'tab' || dir === 'right') {
        return prev.col + 1 < 5
          ? { row: prev.row, col: prev.col + 1 }
          : { row: Math.min(11, prev.row + 1), col: 0 };
      }
      if (dir === 'left') return { row: prev.row, col: Math.max(0, prev.col - 1) };
      if (dir === 'up') return { row: Math.max(0, prev.row - 1), col: prev.col };
      if (dir === 'down') return { row: Math.min(11, prev.row + 1), col: prev.col };
      if (dir === 'enter') return { row: Math.min(11, prev.row + 1), col: 0 };
      return prev;
    });
  };

  const handleExportExcel = async () => {
    const aoa = grid.map((r) => r.map((c) => c.raw));
    const ws = XLSX.utils.aoa_to_sheet(aoa);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, 'Inventario');
    const wbout = XLSX.write(wb, { type: 'base64', bookType: 'xlsx' });
    const uri = FileSystem.cacheDirectory + 'Inventario_2026.xlsx';
    await FileSystem.writeAsStringAsync(uri, wbout, { encoding: FileSystem.EncodingType.Base64 });
    await Sharing.shareAsync(uri);
  };

  const activeCellRef = `${COLS[activeCell.col]}${activeCell.row + 1}`;

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.brand}>VisionSheet Expo</Text>
        <Text style={styles.activeBadge}>Célula: {activeCellRef}</Text>
        <TouchableOpacity style={styles.exportBtn} onPress={handleExportExcel}>
          <Text style={styles.exportBtnText}>Exportar .xlsx</Text>
        </TouchableOpacity>
      </View>

      {/* METADE SUPERIOR: Planilha Excel */}
      <View style={styles.topHalf}>
        <ScrollView horizontal showsHorizontalScrollIndicator={false}>
          <ScrollView>
            {grid.map((row, rIdx) => (
              <View key={rIdx} style={styles.row}>
                <View style={styles.rowNumCell}>
                  <Text style={styles.rowNumText}>{rIdx + 1}</Text>
                </View>
                {row.map((cell, cIdx) => {
                  const isSelected = activeCell.row === rIdx && activeCell.col === cIdx;
                  return (
                    <TouchableOpacity
                      key={cIdx}
                      onPress={() => setActiveCell({ row: rIdx, col: cIdx })}
                      style={[
                        styles.cell,
                        rIdx === 0 && styles.headerCell,
                        isSelected && styles.activeCellBorder,
                      ]}
                    >
                      <Text
                        numberOfLines={1}
                        style={[
                          styles.cellText,
                          cell.bold && { fontWeight: '700' },
                          rIdx === 0 && { color: '#34d399' },
                        ]}
                      >
                        {cell.raw || (isSelected ? `[${activeCellRef}]` : '')}
                      </Text>
                    </TouchableOpacity>
                  );
                })}
              </View>
            ))}
          </ScrollView>
        </ScrollView>
      </View>

      {/* BARRA INTERMEDIÁRIA: fx + Botão Translúcido (*) */}
      <View style={styles.middleBar}>
        <View style={styles.formulaRow}>
          <Text style={styles.fxLabel}>fx</Text>
          <TextInput
            style={styles.fxInput}
            value={grid[activeCell.row][activeCell.col].raw}
            onChangeText={(txt) => {
              setGrid((prev) => {
                const next = prev.map((r) => r.map((c) => ({ ...c })));
                next[activeCell.row][activeCell.col].raw = txt;
                return next;
              });
            }}
            placeholder={`Editar ${activeCellRef} ou =SOMA(...)`}
            placeholderTextColor="#64748b"
          />
          <TouchableOpacity style={styles.translucentToggleBtn} onPress={() => setShowRibbon(!showRibbon)}>
            <Text style={styles.translucentToggleText}>{showRibbon ? '(*) Ocultar' : '(*) Excel & IA'}</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* METADE INFERIOR: Câmera + OCR + Barra Vertical Translúcida (Passo 6) */}
      <View style={styles.bottomHalf}>
        <Animated.View
          {...(isFrozen ? panResponder.panHandlers : {})}
          style={[styles.cameraContainer, { transform: [{ translateX: panX }] }]}
        >
          {!isFrozen ? (
            <TouchableOpacity activeOpacity={0.9} style={styles.liveViewfinder} onPress={handleFreezeFrame}>
              {cameraPermission?.granted && (
                <CameraView ref={cameraRef} style={StyleSheet.absoluteFill} facing="back" enableTorch={torchOn} />
              )}
              <View style={styles.freezePromptBox}>
                <Text style={styles.freezePromptTitle}>Toque para Congelar Frame & Ler OCR</Text>
              </View>
            </TouchableOpacity>
          ) : (
            <View style={styles.frozenSurface}>
              {frozenPhotoUri && <Image source={{ uri: frozenPhotoUri }} style={StyleSheet.absoluteFill} />}
              {INITIAL_OCR_CHIPS.map((chip) => (
                <TouchableOpacity
                  key={chip.id}
                  onPress={() => handleSelectOcrChip(chip)}
                  style={[
                    styles.ocrBoundingBox,
                    { left: `${chip.x}%`, top: `${chip.y}%`, width: `${chip.w}%`, height: `${chip.h}%` },
                  ]}
                >
                  <Text numberOfLines={1} style={styles.ocrText}>{chip.text}</Text>
                </TouchableOpacity>
              ))}
            </View>
          )}
        </Animated.View>

        {/* PASSO 6: Barra Vertical Translúcida na Extrema Direita (TAB, ▲, ►, ◄, ▼, ↵) */}
        <View style={styles.rightVerticalDpad}>
          <TouchableOpacity style={styles.dpadBtn} onPress={() => navigateCell('tab')}>
            <Text style={styles.dpadBtnText}>TAB</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.dpadBtn} onPress={() => navigateCell('up')}>
            <Text style={styles.dpadBtnText}>▲</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.dpadBtn} onPress={() => navigateCell('right')}>
            <Text style={styles.dpadBtnText}>►</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.dpadBtn} onPress={() => navigateCell('left')}>
            <Text style={styles.dpadBtnText}>◄</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.dpadBtn} onPress={() => navigateCell('down')}>
            <Text style={styles.dpadBtnText}>▼</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[styles.dpadBtn, styles.dpadEnterBtn]} onPress={() => navigateCell('enter')}>
            <Text style={[styles.dpadBtnText, { color: '#022c22' }]}>↵</Text>
          </TouchableOpacity>
        </View>

        {statusToast && (
          <View style={styles.toastBanner}>
            <Text style={styles.toastText}>{statusToast}</Text>
          </View>
        )}
      </View>
    </SafeAreaView>
  );
}
